   200→// 创建渐变环境贴图（用于金属反射，兼容所有后端）
   201→function createGradientEnvTexture() {
   202→    const canvas = document.createElement('canvas');
   203→    canvas.width = 256;
   204→    canvas.height = 256;
   205→    const ctx = canvas.getContext('2d');
   206→    const grad = ctx.createLinearGradient(0, 0, 0, 256);
   207→    grad.addColorStop(0, '#1a2550');
   208→    grad.addColorStop(0.5, '#0a1030');
   209→    grad.addColorStop(1, '#050818');
   210→    ctx.fillStyle = grad;
   211→    ctx.fillRect(0, 0, 256, 256);
   212→    const tex = new THREE.CanvasTexture(canvas);
   213→    tex.mapping = THREE.EquirectangularReflectionMapping;
   214→    tex.colorSpace = THREE.SRGBColorSpace;
   215→    return tex;
   216→}

   218→// ===== 加载 GLB 模型 =====
   219→function loadModel() {
   220→    return new Promise((resolve) => {
   221→        const loader = new GLTFLoader();
   222→        loader.load(
   223→            '/scene.glb',
   224→            (gltf) => {
   225→                modelRoot = gltf.scene;
   226→
   227→                // 归一化模型：居中 + 贴地
   228→                normalizeModel(modelRoot);
   229→
   230→                // 遍历设置阴影 + 识别设备
   231→                const groups = { light: null, ac: null, furniture: null, wall: null };
   232→                modelRoot.traverse((child) => {
   233→                    if (child.isMesh) {
   234→                        // 性能优化：只有墙体和大型结构投射阴影，灯具/空调/家具只接收阴影
   235→                        child.receiveShadow = true;
   236→                        child.castShadow = false;
   237→                        // 记录原始材质信息
   238→                        if (child.material) {
   239→                            child.userData.origEmissive = child.material.emissive
   240→                                ? child.material.emissive.clone()
   241→                                : new THREE.Color(0, 0, 0);
   242→                            child.userData.origEmissiveIntensity = child.material.emissiveIntensity || 0;
   243→                            // 确保材质有 UV（避免警告）
   244→                            if (child.geometry && !child.geometry.attributes.uv) {
   245→                                // 为没有 UV 的几何体添加空 UV
   246→                                const posAttr = child.geometry.attributes.position;
   247→                                const uvs = new Float32Array(posAttr.count * 2);
   248→                                child.geometry.setAttribute('uv', new THREE.BufferAttribute(uvs, 2));
   249→                            }
   250→                        }
   251→                    }
   252→                    // 识别中文分组名
   253→                    const n = child.name;
   254→                    if (n === '灯') groups.light = child;
   255→                    else if (n === '空调') groups.ac = child;
   256→                    else if (n === 'zhuoyi' || n === '桌椅') groups.furniture = child;
   257→                    else if (n === 'qiangti' || n === '墙体') groups.wall = child;
   258→                });
   259→
   260→                scene.add(modelRoot);
   261→
   262→                // 按分组识别设备
   263→                if (groups.light) identifyLightFixtures(groups.light);
   264→                if (groups.ac) identifyACUnits(groups.ac);
   265→                if (groups.furniture) identifyFurniture(groups.furniture);
   266→                if (groups.wall) identifyWalls(groups.wall);
   267→                // 墙体投射阴影
   268→                if (groups.wall) {
   269→                    groups.wall.traverse((child) => {
   270→                        if (child.isMesh) child.castShadow = true;
   271→                    });
   272→                }
   273→
   274→                // 设置照明点光源
   275→                setupPointLights();
   276→
   277→                // 设置空调气流
   278→                setupAirflow();
   279→
   280→                // 调整相机适配模型
   281→                fitCameraToModel(modelRoot);
   282→
   283→                // 应用初始状态
   284→                applyLighting();
   285→                applyAC();
   286→
   287→                // 构建 UI
   288→                buildDeviceList();
   289→                updateStatus();
   290→
   291→                console.log(`模型加载完成: ${lightFixtures.length} 个灯具, ${acUnits.length} 个空调`);
   292→                resolve();
   293→            },
   294→            (xhr) => {
   295→                if (xhr.lengthComputable) {
   296→                    const pct = Math.round((xhr.loaded / xhr.total) * 100);
   297→                    updateLoaderText(`加载 3D 模型... ${pct}%`);
   298→                    document.getElementById('loader-bar').style.width = pct + '%';
   299→                }
   300→            },
   301→            (err) => {
   302→                console.error('模型加载失败:', err);
   303→                updateLoaderText('模型加载失败，使用演示场景');
   304→                createDemoScene();
   305→                resolve();
   306→            }
   307→        );
   308→    });
   309→}

   311→// 归一化模型
   312→function normalizeModel(model) {
   313→    const box = new THREE.Box3().setFromObject(model);
   314→    const size = box.getSize(new THREE.Vector3());
   315→    const center = box.getCenter(new THREE.Vector3());
   316→    console.log('原始模型包围盒:', `size=${size.x.toFixed(2)}x${size.y.toFixed(2)}x${size.z.toFixed(2)}`, `center=(${center.x.toFixed(2)},${center.y.toFixed(2)},${center.z.toFixed(2)})`);
   317→
   318→    // 归一化到约 40 单位大小
   319→    const maxDim = Math.max(size.x, size.y, size.z);
   320→    const scale = 40 / maxDim;
   321→    model.scale.setScalar(scale);
   322→
   323→    // 重新计算 box 并居中（XZ），贴地（Y=0）
   324→    const box2 = new THREE.Box3().setFromObject(model);
   325→    const center2 = box2.getCenter(new THREE.Vector3());
   326→    model.position.x -= center2.x;
   327→    model.position.z -= center2.z;
   328→    model.position.y -= box2.min.y;
   329→
   330→    const box3 = new THREE.Box3().setFromObject(model);
   331→    console.log('归一化后包围盒:', `min=(${box3.min.x.toFixed(2)},${box3.min.y.toFixed(2)},${box3.min.z.toFixed(2)})`, `max=(${box3.max.x.toFixed(2)},${box3.max.y.toFixed(2)},${box3.max.z.toFixed(2)})`);
   332→}

   351→// 识别空调
   352→function identifyACUnits(groupNode) {
   353→    // 直接子节点作为独立空调单元
   354→    const directChildren = groupNode.children.filter(c => c.isMesh || c.isObject3D);
   355→    directChildren.forEach((child, idx) => {
   356→        // 收集该子节点下所有 mesh
   357→        const meshes = [];
   358→        child.traverse((m) => {
   359→            if (m.isMesh) meshes.push(m);
   360→        });
   361→        if (meshes.length === 0 && child.isMesh) meshes.push(child);
   362→        if (meshes.length === 0) return;
   363→
   364→        const acInfo = makeDeviceInfo(child, 'ac', idx);
   365→        meshes.forEach((m) => {
   366→            m.userData.deviceType = 'ac';
   367→            m.userData.deviceInfo = acInfo;
   368→            m.userData.acIndex = idx;
   369→            acUnits.push(m);
   370→            selectableObjects.push(m);
   371→        });
   372→    });
   373→}

   397→// 生成设备信息（带编号）
   398→function makeDeviceInfo(obj, type, index = 0) {
   399→    const base = deviceDB[type] || deviceDB.default;

   406→        ...base,

   409→        size: `${size.x.toFixed(1)} × ${size.y.toFixed(1)} × ${size.z.toFixed(1)}`,

   411→        power: type === 'light' ? `${base.power}W` : type === 'ac' ? `${base.power}W` : '-'

   415→// ===== 设置点光源（从灯具位置采样） =====
   416→function setupPointLights() {
   417→    // 收集所有灯具的世界坐标
   418→    const positions = [];
   419→    const tmp = new THREE.Vector3();
   420→    lightFixtures.forEach((mesh) => {
   421→        mesh.getWorldPosition(tmp);
   422→        positions.push(tmp.clone());
   423→    });
   424→
   425→    if (positions.length === 0) return;
   426→
   427→    // 用最远点采样选出最多 N 个代表性位置
   428→    const maxLights = 8;
   429→    const samples = farthestPointSampling(positions, Math.min(maxLights, positions.length));
   430→
   431→    samples.forEach((pos, i) => {
   432→        const light = new THREE.PointLight(0xffe8b0, 0, 25, 1.5);
   433→        light.position.set(pos.x, pos.y - 0.3, pos.z);
   434→        light.name = `__pointLight_${i}`;
   435→        scene.add(light);
   436→        pointLights.push(light);
   437→    });
   438→}
   439→
   440→// 最远点采样算法
   441→function farthestPointSampling(points, n) {
   442→    if (points.length <= n) return points.slice();
   443→    const result = [points[0]];
   444→    const remaining = points.slice(1);
   445→    while (result.length < n && remaining.length > 0) {
   446→        let maxDist = -1;
   447→        let maxIdx = 0;
   448→        for (let i = 0; i < remaining.length; i++) {
   449→            let minDist = Infinity;
   450→            for (const r of result) {
   451→                const d = remaining[i].distanceTo(r);
   452→                if (d < minDist) minDist = d;
   453→            }
   454→            if (minDist > maxDist) {
   455→                maxDist = minDist;
   456→                maxIdx = i;
   457→            }
   458→        }
   459→        result.push(remaining[maxIdx]);
   460→        remaining.splice(maxIdx, 1);
   461→    }
   462→    return result;
   463→}

   465→// ===== 设置空调气流粒子 =====
   466→function setupAirflow() {
   467→    // 按空调索引分组，每组创建一个气流系统
   468→    const groupMap = new Map();
   469→    acUnits.forEach((mesh) => {
   470→        const idx = mesh.userData.acIndex ?? 0;
   471→        if (!groupMap.has(idx)) groupMap.set(idx, []);
   472→        groupMap.get(idx).push(mesh);
   473→    });
   474→
   475→    groupMap.forEach((meshes, idx) => {
   476→        // 计算该组中心
   477→        const box = new THREE.Box3();
   478→        meshes.forEach((m) => box.expandByObject(m));
   479→        const center = box.getCenter(new THREE.Vector3());
   480→        const size = box.getSize(new THREE.Vector3());
   481→
   482→        const system = createAirflowSystem(center, size, idx);
   483→        airflowSystems.push(system);
   484→        scene.add(system.points);
   485→    });
   486→}

   488→// 创建单个气流粒子系统
   489→function createAirflowSystem(center, size, index) {
   490→    const count = 120;
   491→    const positions = new Float32Array(count * 3);
   492→    const velocities = new Float32Array(count * 3);
   493→    const ages = new Float32Array(count);
   494→    const lifetimes = new Float32Array(count);
   495→
   496→    for (let i = 0; i < count; i++) {
   497→        resetAirflowParticle(positions, velocities, ages, lifetimes, i, center, size);
   498→    }
   499→
   500→    const geo = new THREE.BufferGeometry();
   501→    geo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
   502→
   503→    // 圆形粒子贴图
   504→    const tex = createCircleTexture();
   505→
   506→    const mat = new THREE.PointsMaterial({
   507→        color: 0x88ccff,
   508→        size: 0.35,
   509→        map: tex,
   510→        transparent: true,
   511→        opacity: 0.55,
   512→        depthWrite: false,
   513→        blending: THREE.AdditiveBlending,
   514→        sizeAttenuation: true
   515→    });
   516→
   517→    const points = new THREE.Points(geo, mat);
   518→    points.userData.isAirflow = true;
   519→    points.userData.center = center.clone();
   520→    points.userData.size = size.clone();
   521→    points.userData.velocities = velocities;
   522→    points.userData.ages = ages;
   523→    points.userData.lifetimes = lifetimes;
   524→    points.userData.acIndex = index;
   525→
   526→    return { points, mat };
   527→}

   529→function resetAirflowParticle(positions, velocities, ages, lifetimes, i, center, size) {
   530→    // 从空调出风口位置出发
   531→    positions[i * 3] = center.x + (Math.random() - 0.5) * size.x * 0.8;
   532→    positions[i * 3 + 1] = center.y;
   533→    positions[i * 3 + 2] = center.z + (Math.random() - 0.5) * size.z * 0.8;
   534→
   535→    // 向下 + 随机扩散
   536→    const angle = Math.random() * Math.PI * 2;
   537→    const spread = 0.3 + Math.random() * 0.4;
   538→    velocities[i * 3] = Math.cos(angle) * spread;
   539→    velocities[i * 3 + 1] = -(0.6 + Math.random() * 0.8);
   540→    velocities[i * 3 + 2] = Math.sin(angle) * spread;
   541→
   542→    ages[i] = 0;
   543→    lifetimes[i] = 2.5 + Math.random() * 2;
   544→}

   569→    dist *= 1.4;

   571→    initialCamera.pos.set(dist * 0.8, dist * 0.55, dist * 0.8);

   573→
   574→    camera.position.copy(initialCamera.pos);
   575→    controls.target.copy(initialCamera.target);
   576→    controls.update();

   642→function getMouseNDC(event) {

   644→    return new THREE.Vector2(
   645→        ((event.clientX - rect.left) / rect.width) * 2 - 1,

   652→function onPointerMove(event) {
   653→    const mouse = getMouseNDC(event);
   654→    raycaster.setFromCamera(mouse, camera);
   655→    const hits = raycaster.intersectObjects(selectableObjects, false);
   656→
   657→    if (hits.length > 0) {
   658→        const obj = hits[0].object;
   659→        if (hoveredObject !== obj) {
   660→            hoveredObject = obj;
   661→            renderer.domElement.style.cursor = 'pointer';
   662→        }
   663→    } else {
   664→        if (hoveredObject) {
   665→            hoveredObject = null;
   666→            renderer.domElement.style.cursor = 'grab';
   667→        }
   668→    }
   669→}

   671→function onPointerClick(event) {
   672→    const mouse = getMouseNDC(event);
   673→    raycaster.setFromCamera(mouse, camera);
   674→    const hits = raycaster.intersectObjects(selectableObjects, false);
   675→
   676→    if (hits.length > 0) {
   677→        const obj = findSelectableRoot(hits[0].object);
   678→        if (selectedObject === obj) {
   679→            deselectObject();
   680→        } else {
   681→            deselectObject();
   682→            selectObject(obj);
   683→        }
   684→    } else {
   685→        deselectObject();
   686→    }
   687→}

   689→function onDoubleClick(event) {
   690→    const mouse = getMouseNDC(event);
   691→    raycaster.setFromCamera(mouse, camera);
   692→    const hits = raycaster.intersectObjects(selectableObjects, false);
   693→    if (hits.length > 0) {
   694→        const obj = findSelectableRoot(hits[0].object);
   695→        focusOnObject(obj);
   696→        selectObject(obj);
   697→    }
   698→}

   886→// ===== 空调系统 =====
   887→function applyAC() {
   888→    const { enabled, temperature } = state.ac;
   889→
   890→    // 气流粒子可见性
   891→    airflowSystems.forEach((sys) => {
   892→        sys.mat.opacity = enabled ? 0.55 : 0;
   893→        sys.mat.visible = enabled;
   894→    });
   895→
   896→    // 温度色调映射：冷色(16°C) -> 中性(24°C) -> 暖色(30°C)
   897→    const t = (temperature - 16) / 14; // 0-1
   898→    let fogColor, ambientColor, particleColor;
   899→
   900→    if (t < 0.4) {
   901→        // 冷色调：蓝
   902→        const k = (0.4 - t) / 0.4;
   903→        fogColor = new THREE.Color(0x06081a).lerp(new THREE.Color(0x0a1428), k);
   904→        ambientColor = new THREE.Color(0x6688aa).lerp(new THREE.Color(0x4466aa), k * 0.5);
   905→        particleColor = new THREE.Color(0x88ccff).lerp(new THREE.Color(0x44aaff), k * 0.5);
   906→    } else if (t > 0.6) {
   907→        // 暖色调：橙
   908→        const k = (t - 0.6) / 0.4;
   909→        fogColor = new THREE.Color(0x06081a).lerp(new THREE.Color(0x1a1008), k);
   910→        ambientColor = new THREE.Color(0x6688aa).lerp(new THREE.Color(0xaa8866), k * 0.5);
   911→        particleColor = new THREE.Color(0x88ccff).lerp(new THREE.Color(0xffaa66), k * 0.6);
   912→    } else {
   913→        // 中性
   914→        fogColor = new THREE.Color(0x06081a);
   915→        ambientColor = new THREE.Color(0x6688aa);
   916→        particleColor = new THREE.Color(0x88ccff);
   917→    }
   918→
   919→    if (enabled) {
   920→        scene.fog.color = fogColor;
   921→        scene.background = fogColor;
   922→    } else {
   923→        scene.fog.color = new THREE.Color(0x06081a);
   924→        scene.background = new THREE.Color(0x06081a);
   925→    }
   926→
   927→    const ambient = scene.getObjectByName('__ambient');
   928→    if (ambient && state.lighting.enabled) {
   929→        ambient.color = enabled ? ambientColor : new THREE.Color(0x6688aa);
   930→    }
   931→
   932→    // 气流粒子颜色
   933→    airflowSystems.forEach((sys) => {
   934→        sys.mat.color = enabled ? particleColor : sys.mat.color;
   935→    });
   936→
   937→    // 更新空调设备信息中的温度
   938→    acUnits.forEach((mesh) => {
   939→        if (mesh.userData.deviceInfo) {
   940→            mesh.userData.deviceInfo.temp = enabled ? `${temperature}°C` : '待机';
   941→        }
   942→    });
   943→    if (selectedObject && selectedObject.userData.deviceType === 'ac') {
   944→        showDeviceInfo(selectedObject);
   945→    }
   946→
   947→    // UI
   948→    const tag = document.getElementById('ac-status');
   949→    tag.textContent = enabled ? '运行中' : '已关闭';
   950→    tag.classList.toggle('off', !enabled);
   951→    document.getElementById('ac-toggle').classList.toggle('active', enabled);
   952→    document.getElementById('ac-toggle-text').textContent = enabled ? '空调已开启' : '空调已关闭';
   953→    document.getElementById('temperature-value').textContent = temperature;
   954→
   955→    updateStatus();
   956→}

   958→// ===== 状态栏更新 =====
   959→function updateStatus() {
   960→    const lightCount = lightFixtures.length > 0 ? new Set(lightFixtures.map(m => m.userData.deviceInfo?.name)).size : 0;
   961→    const acCount = acUnits.length > 0 ? new Set(acUnits.map(m => m.userData.acIndex)).size : 0;
   962→    const total = lightCount + acCount;
   963→    const active = (state.lighting.enabled ? lightCount : 0) + (state.ac.enabled ? acCount : 0);
   964→
   965→    const lightPower = state.lighting.enabled ? lightCount * deviceDB.light.power : 0;
   966→    const acPower = state.ac.enabled ? acCount * deviceDB.ac.power : 0;
   967→    const totalPower = (lightPower + acPower) / 1000;
   968→
   969→    document.getElementById('total-devices').innerHTML = `${total}<span class="unit">台</span>`;
   970→    document.getElementById('active-devices').innerHTML = `${active}<span class="unit">台</span>`;
   971→    document.getElementById('total-power').innerHTML = `${totalPower.toFixed(1)}<span class="unit">kW</span>`;
   972→    document.getElementById('avg-temp').innerHTML = `${state.ac.enabled ? state.ac.temperature : '--'}<span class="unit">°C</span>`;
   973→    document.getElementById('lux-value').innerHTML = `${state.lighting.enabled ? Math.round(300 + state.lighting.brightness * 450) : 30}<span class="unit">lx</span>`;
   974→
   975→    // 能耗等级
   976→    let level = 'A';
   977→    if (totalPower > 8) level = 'C';
   978→    else if (totalPower > 5) level = 'B';
   979→    else if (totalPower > 2) level = 'A';
   980→    else level = 'A+';
   981→    document.getElementById('energy-level').textContent = level;
   982→}

  1180→// ===== 动画循环 =====
  1181→function animate() {
  1182→    requestAnimationFrame(animate);
  1183→    const delta = clock.getDelta();
  1184→
  1185→    // 相机动画
  1186→    if (cameraTween) {
  1187→        cameraTween.time += delta;
  1188→        const t = Math.min(cameraTween.time / cameraTween.duration, 1);
  1189→        const eased = t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
  1190→        camera.position.lerpVectors(cameraTween.startPos, cameraTween.endPos, eased);
  1191→        controls.target.lerpVectors(cameraTween.startTarget, cameraTween.endTarget, eased);
  1192→        if (t >= 1) cameraTween = null;
  1193→    }
  1194→
  1195→    controls.update();
  1196→
  1197→    // 更新气流粒子
  1198→    if (state.ac.enabled) {
  1199→        updateAirflow(delta);
  1200→    }
  1201→
  1202→    // FPS
  1203→    fpsCounter.frames++;
  1204→    const now = performance.now();
  1205→    if (now - fpsCounter.lastTime >= 1000) {
  1206→        fpsCounter.value = Math.round((fpsCounter.frames * 1000) / (now - fpsCounter.lastTime));
  1207→        fpsCounter.frames = 0;
  1208→        fpsCounter.lastTime = now;
  1209→        document.getElementById('fps-value').textContent = fpsCounter.value;
  1210→    }
  1211→
  1212→    renderer.render(scene, camera);
  1213→}

  1215→// 更新气流粒子
  1216→function updateAirflow(delta) {
  1217→    airflowSystems.forEach((sys) => {
  1218→        const points = sys.points;
  1219→        const pos = points.geometry.attributes.position.array;
  1220→        const vel = points.userData.velocities;
  1221→        const ages = points.userData.ages;
  1222→        const lifetimes = points.userData.lifetimes;
  1223→        const center = points.userData.center;
  1224→        const size = points.userData.size;
  1225→
  1226→        for (let i = 0; i < pos.length / 3; i++) {
  1227→            ages[i] += delta;
  1228→            if (ages[i] > lifetimes[i] || pos[i * 3 + 1] < 0) {
  1229→                resetAirflowParticle(pos, vel, ages, lifetimes, i, center, size);
  1230→                continue;
  1231→            }
  1232→            pos[i * 3] += vel[i * 3] * delta;
  1233→            pos[i * 3 + 1] += vel[i * 3 + 1] * delta;
  1234→            pos[i * 3 + 2] += vel[i * 3 + 2] * delta;
  1235→            // 重力衰减 + 扩散
  1236→            vel[i * 3 + 1] += 0.3 * delta;
  1237→            vel[i * 3] *= 1.02;
  1238→            vel[i * 3 + 2] *= 1.02;
  1239→        }
  1240→        points.geometry.attributes.position.needsUpdate = true;
  1241→    });
  1242→}
